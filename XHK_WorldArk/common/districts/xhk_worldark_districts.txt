	base_buildtime = @xhk_worldark_city_time
	icon = district_city
	is_capped_by_modifier = no
	
	planet_modifier = {
		planet_housing_add = @xhk_worldark_city_housing
	}

    inline_script = {
        script = xhk_worldark/xhk_worldark_capital_part
        OWNER_REQ = "is_gestalt = no"
    }

	inline_script = {
        script = xhk_worldark/xhk_worldark_housing_modifiers
        BASE = 100
        HOUSING2 = 200
    }

    # tankbound
    triggered_planet_modifier = {
        potential = {
            exists = owner
            owner = {
                is_tankbound_empire = yes
            }
        }
        modifier = {
            job_production_overseer_add = 100
        }
    }
}

district_xhk_worldark_foundry = {
    base_buildtime = @xhk_worldark_city_time
    icon = district_arcology_arms_industry
    gridbox = district_arcology_arms_industry
    is_capped_by_modifier = no
    overlay_icon = GFX_district_alloys

    zone_slots = {
        slot_xhk_worldark_foundry
    }

    potential = {
        always = yes
    }

    resources = {
        category = planet_districts_industrial
        cost = {
            minerals = @xhk_worldark_city_cost
        }
        upkeep = {
            energy = @xhk_worldark_city_upkeep
        }

        # nanite upgrade
        upkeep = {
            trigger = {
                frameworld_has_active_nanite_upgrade = yes
            }
            energy = 10
            nanites = 1
        }
        produces = {
            trigger = {
                frameworld_has_active_nanite_upgrade = yes
            }
            alloys = 5
        }
    }

    planet_modifier = {
        planet_housing_add = @xhk_worldark_standard_housing
    }

    inline_script = {
        script = jobs/foundry_add
        AMOUNT = 300
    }
}
