xhk_worldark_size_harvest = {
    user_scope = fleet
    scope = planet

	automation = {
		priority	= 2
		default_on	= yes
		tooltip		= xhk_worldark_size_harvest_automate_tt
		options = {
			inline_script = scripted_actions/harvest_automation_options
		}
	}

    on_completed = xhk_on_size_harvest
    tooltip = xhk_worldark_size_harvest_ship_button
    icon = GFX_XHK_worldark_size_harvest_action
    icon_selected = GFX_XHK_worldark_size_harvest_action

    possible = {
        NOT = {
            has_modifier = "xhk_size_harvested_already"
        }
        if = {
            limit = {
                is_surveyed = {
                    who = from
                    status = yes
                }
                has_anomaly = yes
            }
            custom_tooltip = {
                fail_text = arkship_target_has_anomaly_fail_tt
                always = no
            }
        }
        xhk_worldark_is_valid_size_harvest_target = yes
    }

    button_visible = {
        custom_tooltip = {
            fail_text = xhk_worldark_dummy_error
            starbase? = {
                xhk_is_worldark_starbase = yes
            }
        }
    }

    button_clickable = {
        starbase? = {
            xhk_is_worldark_starbase = yes
        }
        # starbase.ship.carrier = {
        #     NOT = { has_modifier = xhk_worldark_harvest_size_cooldown }
        # }
    }

    activity_key = xhk_worldark_size_harvest_activity_key
    context_menu_name = xhk_worldark_size_harvest
    required_progress = value:arkship_harvest_resources_required_progress_value
    progress_activity_key = xhk_worldark_size_harvest_progress_activity_key
    required_progress_key = xhk_worldark_size_harvest_progress_progress_key
    cooldown = @xhk_worldark_harvest_cd
    # category = harvest_resources
    # category_limit = 1

    uses_vfx_type = harvest_effect

    cost = {
        energy = @arkship_strip_mine_cost
    }
    slot = 3

    completion_message_type = XHK_TOAST_HARVEST_SIZE_COMPLETE
    completion_message_localization = XHK_TOAST_HARVEST_SIZE_COMPLETE_DESC

    ai = {
        #This - Target Planet
        #From - Acting Fleet
        weight = {
            base = 1000 # ME FIRST
            modifier = {
                factor = 0 #cannot afford
                FROM.controller = {
                    resource_stockpile_compare = {
                        resource = energy
                        value < @arkship_strip_mine_cost
                        mult = 1.5
                    }
                }
            }
            modifier = {
                factor = 0
                from = { has_fleet_flag = xhk_expand_harvest_cooldown }
            }
            inline_script = {
                script = scripted_actions/ai_distance_modifiers
                LOCAL_SYSTEM_CHECK = "any_system_planet = { is_same_value = ROOT }"
            }
        }
        max_jumps = 4
        good_enough_weight = 15
        ai_minister = interior
    }
}

xhk_worldark_corespinal_planeteater_dismantle_world = {
	user_scope = fleet
	scope = planet

	on_progress_start = xhk_onaction_corespinal_planeteater
	on_cancelled = xhk_onaction_corespinal_planeteater_cancelled
	tooltip = xhk_worldark_corespinal_planeteater_dismantle_world_ship_button
	icon = GFX_fleet_action_button_arkship_strip_mine
	icon_selected = GFX_fleet_action_button_arkship_strip_mine

	important = yes

	slot = 2

	possible = {
		custom_tooltip = {
			fail_text = xhk_worldark_is_star_fail_tt
			is_star = no
		}
		custom_tooltip = {
			fail_text = xhk_worldark_is_shielded_fail_tt
			NOT = { is_planet_class = pc_shielded }
		}
		custom_tooltip = {
			fail_text = arkship_lithoid_consume_world_uninhabited_fail_tt
			is_colony = no
		}
		custom_tooltip = {
			fail_text = arkship_target_not_surveyed_fail_tt
			is_surveyed = {
				who = from
				status = yes
			}
		}
		custom_tooltip = {
			fail_text = xhk_worldark_corespinal_planeteater_dismantle_world_being_devoured_fail_tt
			OR = {
				NOT = { has_carrier_flag = being_devoured }
				AND = {
					any_targeting_situation = {
						is_situation_type = xhk_worldark_situation_planeteater
						is_owned_by = from.controller
					}
					OR = {
						from = { has_fleet_flag = consuming_world_arkship@root }
						has_carrier_flag = arkship_absent_grace
					}
				}
			}
		}
	}

	button_visible = { always = no }

	hide_from_context_menu = {
		always = no
		# from.starbase? = {
		# 	NOT = { has_starbase_module = xhk_worldark_corespinal_planeteater }
		# }
	}

	uses_vfx_type = harvest_effect
	activity_key = xhk_worldark_corespinal_planeteater_dismantle_world_activity_key
	context_menu_name = xhk_worldark_corespinal_planeteater_dismantle_world
	required_progress = @arkship_consume_world_required_progress
	progress_activity_key = xhk_worldark_corespinal_planeteater_dismantle_world_progress_activity_key
	required_progress_key = xhk_worldark_corespinal_planeteater_dismantle_world_required_progress_key

	ai = {
		#This - Target Planet
		#From - Acting Fleet
		weight = {
			base = @ai_action_weight_25
			modifier = {
				factor = 0 #already consuming a planet
				FROM.controller = {
					any_situation = {
						is_situation_type = situation_arkship_lithoid_consume_planet
					}
				}
			}
			inline_script = {
				script = scripted_actions/ai_distance_modifiers
				LOCAL_SYSTEM_CHECK = "any_system_planet = { is_same_value = ROOT }"
			}
			modifier = {
				factor = @ai_action_weight_factor_2 #prefer unowned systems
				NOT = { exists = space_owner }
			}
			modifier = {
				factor = @ai_action_weight_factor_0_1 #avoid owned systems
				exists = space_owner
			}
		}
		max_jumps = @ai_action_weight_10
		good_enough_weight = @ai_action_weight_30
		ai_minister = interior
	}
}
