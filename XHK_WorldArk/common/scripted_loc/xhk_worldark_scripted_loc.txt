# harvest tt
defined_text = {
    name = xhk_worldark_getDistrictIncreaseFromHarvest
	text = {
		trigger = {
			capital_scope.ship.starbase = {
				has_starbase_size >= xhk_worldark_tier_11
			}
		}
		localization_key = xhk_harvest_plus_vlarge
	}
    text = {
        trigger = {
			capital_scope.ship.starbase = {
				has_starbase_size >= xhk_worldark_tier_5
				NOT = { has_starbase_size >= xhk_worldark_tier_11  }
			}
		}
        localization_key = xhk_harvest_plus_large
    }
    default = xhk_harvest_plus_small
}
