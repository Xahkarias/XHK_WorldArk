# traders
defined_text = {
	name = GetBaseTrader
	text = {
		trigger = {
			OR = {
				AND = {
					NOT = {
						is_scope_type = country
					}
					exists = owner
					owner = {
						is_gestalt = yes
					}
				}
				AND = {
					is_scope_type = country
					is_gestalt = yes
				}
			}
		}
		localization_key = job_logistics_drone
	}
	text = {
		trigger = {
			OR = {
				AND = {
					NOT = {
						is_scope_type = country
					}
					exists = owner
					owner = {
						is_gestalt = no
					}
				}
				AND = {
					is_scope_type = country
					is_gestalt = no
				}
			}
		}
		localization_key = job_trader
	}
	default = job_trader
}
