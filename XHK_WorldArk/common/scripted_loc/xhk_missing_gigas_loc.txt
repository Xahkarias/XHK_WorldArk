define_text = {
	name = get_dark_matter_weaver
	text = {
		trigger = {
			OR = {
				AND = {
					NOT = {
						is_scope_type = country
					}
					exists = owner
					owner = {
						is_gestalt = yes
					}
				}
				AND = {
					is_scope_type = country
					is_gestalt = yes
				}
			}
		}
		localization_key = job_giga_dark_matter_weaver_drone
	}
	default = job_giga_dark_matter_weaver
}
