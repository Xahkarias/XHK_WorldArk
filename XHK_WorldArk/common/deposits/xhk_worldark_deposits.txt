d_xhk_worldark_base = {
    is_for_colonizable = yes
    icon = d_frameworld_core
    category = deposit_cat_rare
    use_for_min_max_adjustments = yes

    drop_weight = { weight = 0 }
    potential = { always = no }

    planet_modifier = {
        pop_environment_tolerance = 1.0
    }

    ########################################################################################################
    # DISTRICT CONTROL
    ########################################################################################################

    # dystopian limit
    triggered_planet_modifier = {
        potential = {
            owner? = {
                is_regular_empire = yes
                has_valid_civic = civic_dystopian_society
            }
        }
        district_xhk_worldark_amenities_dystopian_max_add = 0.0001
        multiplier = trigger:pop_amount
    }

	########################################################################################################
	# VIRTUAL NERFS
	########################################################################################################
	# virtual colony size
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				or = {
					is_machine_empire = yes
					is_individual_machine = yes
				}
			}
		}
		giga_virtual_colony_size = 1
		multiplier = value:xhk_worldark_level_virtual_colony_size
	}

	########################################################################################################
	# ADDING ARKSHIP TYPE BUFFS
	########################################################################################################
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				xhk_has_worldark_origin = yes
			}
		}
		ship_sensor_range_add = 1
		ship_hyperlane_range_add = 1
		science_ship_survey_speed = 0.05
		multiplier = xhk_worldark_science_upgrade_level
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				xhk_has_worldark_origin = yes
			}
		}
		ship_fire_rate_mult = 0.05
		ship_weapon_range_mult = 0.05
		multiplier = xhk_worldark_military_upgrade_level
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				xhk_has_worldark_origin = yes
			}
		}
		arkship_harvest_resources_produces_mult = 0.05
		multiplier = xhk_worldark_civilian_upgrade_level
	}

	# logistic
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				xhk_has_worldark_origin = yes
			}
		}
		ship_starbase_stockpile_collection_rate_add = 200
		multiplier = xhk_worldark_logistics
	}
}

            xhk_worldark_has_cloning_assembly_facilities = yes
