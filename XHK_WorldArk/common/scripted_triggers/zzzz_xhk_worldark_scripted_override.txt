# Planet-scope allow check: true when the matching $RING$ / $ARKSHIP$ starbase variant is absent.
orbital_ring_and_arkship_equivalent_absent = {
	if = {
		limit = {
			or = {
				is_planet_class = pc_ark
				is_planet_class = xhk_pc_worldark
			}
		}
		fleet.starbase? = {
			NOR = {
				has_starbase_building = $ARKSHIP$
				is_starbase_building_building = $ARKSHIP$
			}
		}
	}
	else = {
		orbital_defence.starbase? = {
			NOR = {
				has_starbase_building = $RING$
				is_starbase_building_building = $RING$
			}
		}
	}
}
