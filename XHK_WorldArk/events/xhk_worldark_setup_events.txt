namespace = xhk_worldark_setup

# INITIAL SETUP
country_event = {
    id = xhk_worldark_setup.1
    hide_window = yes
    fire_only_once = yes
    is_triggered_only = yes

    trigger = {
        xhk_has_worldark_origin = yes
    }

    immediate = {
        give_technology = {
            tech = xhk_tech_worldark_1
            message = no
        }

        # old capital scope
        capital_scope = {
            solar_system = {
                save_event_target_as = target_system
            }
            save_event_target_as = xhk_old_arkcap_colony

            ship = {
                save_event_target_as = xhk_old_arkcap_ship
                fleet = { save_event_target_as = xhk_old_arkcap_fleet }

                # make the new ship
                create_fleet = {
                    effect = {
                        set_owner = root
                        create_ship = {
                            name = random
                            random_existing_design = xhk_worldark_tier_1
                            graphical_culture = xhk_worldark
                            create_colony = yes
                            effect = {
                                save_event_target_as = xhk_new_worldark_ship
                            }
                        }
                        set_location = event_target:target_system
                    }
                }

                # # grab the governor
                colony = {
                    save_event_target_as = xhk_new_worldark_colony
                    leader = {
                        unassign_leader = this
                        save_event_target_as = xhk_worldark_governor
                    }
                }

                # grab the admiral
                fleet = {
                    leader = {
                        unassign_leader = this
                        save_event_target_as = xhk_worldark_admiral
                    }
                }


                event_target:xhk_new_worldark_ship = {
                    # put leaders in new ship
                    colony = {
                        assign_leader = event_target:xhk_worldark_governor

                        set_capital = yes
                    }
                    fleet = {
                        assign_leader = event_target:xhk_worldark_admiral
                        # set the fleet name
                        set_name = {
                            key = "xhk_worldark_new_fleet_name"
                            variable_string = "[xhk_old_arkcap_fleet.GetName]"
                        }
                    }

                    #put in the old module / systems
                    initialize_arkship_starbase_effect = yes

                    set_name = {
                        key = "xhk_worldark_new_capital_name"
                        variable_string = "[xhk_old_arkcap_colony.GetName]"
                    }
                }
            }

            #move the pops
            # this is from the old capital scope to the new one
            every_owned_pop_group = {
                resettle_pop_group = {
                    POP_GROUP = this
                    PLANET = root.capital_scope
                    PERCENTAGE = 100
                }
            }
        }

        # new capital scope
        capital_scope = {

            # HAVE TO DO THIS SHIT MANUALLY
            ###################################################

            # basic resource init
            set_variable = {
                which = energy_districts
                value = 2
            }
            set_variable = {
                which = mining_districts
                value = 2
            }
            set_variable = {
                which = farming_districts
                value = 3
            }


            # from authority regular nomad
            if = {
                limit = {
                    owner? = {
                        is_regular_empire = yes
                        is_nomadic = yes
                    }
                }
                add_district = district_xhk_worldark_city
                add_district = district_xhk_worldark_city
                add_zone = {
                    district = district_xhk_worldark_city
                    zone = zone_ark_research_unity
                    zone_slot = 1
                }
                add_zone = {
                    district = district_xhk_worldark_city
                    zone = zone_ark_industrial
                    zone_slot = 2
                }
                add_building = {
                    building = building_factory_1
                    zone = zone_ark_industrial
                }
            }
            # from authorithy hive nomad
            if = {
                limit = {
                    owner? = {
                        is_hive_empire = yes
                        is_wilderness_empire = no
                        is_nomadic = yes
                    }
                }
                add_building = building_hive_capital
                if = {
                    limit = {
                        owner = {
                            NOT = { has_origin = origin_progenitor_hive }
                        }
                    }
                    add_building = building_spawning_pool
                }
                add_building = building_hive_warren

                add_district = district_xhk_worldark_city
                add_district = district_xhk_worldark_city

                add_zone = {
                    district = district_xhk_worldark_city
                    zone = zone_ark_research_unity_gestalt
                    zone_slot = 1
                }
                add_zone = {
                    district = district_xhk_worldark_city
                    zone = zone_ark_foundry
                    zone_slot = 2
                }
            }
            # from machine auth
            if = {
                limit = {
                    owner? = {
                        is_machine_empire = yes
                        is_nomadic = yes
                    }
                }
                add_building = building_machine_capital
                add_building = building_machine_assembly_plant
                add_building = building_drone_storage

                add_district = district_xhk_worldark_city
                add_district = district_xhk_worldark_city

                add_zone = {
                    district = district_xhk_worldark_city
                    zone = zone_ark_research_unity_gestalt
                    zone_slot = 1
                }

                if = {
                    limit = {
                        owner = {
                            OR = {
                                has_valid_civic = civic_machine_servitor
                                has_valid_civic = civic_machine_obsessional_directive
                            }
                        }
                    }
                    add_zone = {
                        district = district_xhk_worldark_city
                        zone = zone_ark_industrial
                        zone_slot = 2
                    }
                    add_building = {
                        building = building_factory_1
                        zone = zone_ark_industrial
                    }
                }
                else = {
                    add_zone = {
                        district = district_xhk_worldark_city
                        zone = zone_ark_foundry
                        zone_slot = 2
                    }
                }


                set_variable = {
                    which = energy_districts
                    value = 3
                }
                set_variable = {
                    which = mining_districts
                    value = 4
                }
                set_variable = {
                    which = farming_districts
                    value = 0
                }
            }

            ## Species
            inline_script = "game_start/species_lithoids"
            inline_script = "game_start/species_infernals"
            inline_script = "game_start/species_individual_machine"
            inline_script = "game_start/species_subspecies" ## Set subspecies flags

            # Shipsets
            inline_script = "game_start/shipsets"

            ## Ethic
            inline_script = "game_start/ethic_spiritualist"
            inline_script = "game_start/ethic_materialist"
            inline_script = "game_start/ethic_pacifist"
            inline_script = "game_start/ethic_militarist"
            inline_script = "game_start/ethic_egalitarian"
            inline_script = "game_start/ethic_authoritarian"
            inline_script = "game_start/ethic_xenophile"
            inline_script = "game_start/ethic_xenophobe"

            ##Civics
            inline_script = "game_start/civics"

            # Tankbound civic (removes every farming districts so we want to do that at the end)
            inline_script = "game_start/civic_tankbound"

            # energy
            if = {
                limit = {
                    is_planet_class = pc_ark
                }
                if = {
                    limit = {
                        check_variable = {
                            which = energy_districts
                            value > 2
                        }
                    }
                    set_variable = {
                        which = energy_districts
                        value = 2
                    }
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = energy_districts
                        value >= 1
                    }
                }
                add_district_and_planet_size_if_needed_effect_amount = {
                    amount = energy_districts
                    district = district_xhk_worldark_generator
                }
            }
            # food or minerals
            if = {
                limit = {
                    check_variable = {
                        which = farming_districts
                        value >= 1
                    }
                }
                add_district = district_xhk_worldark_farming
                add_zone = {
                    district = district_xhk_worldark_farming
                    zone = zone_xhk_worldark_farming
                }
            }
            else = {
                add_district = district_xhk_worldark_minerals
            }

            # Clear variables and flags
            clear_variable = energy_districts
            clear_variable = mining_districts
            clear_variable = farming_districts

            # AT THE END IF NEEDED
            upgrade_colony_shelter = yes

            #kill the extra pops made by moving the ship
            while = {
                count = 10
                random_owned_pop_group = {
                    limit = {
                        is_same_species = owner.species
                    }
                    kill_single_pop = yes
                }
            }


            # control deposit
            add_deposit = d_xhk_worldark_base
        }

        # homeworld fix
        species = {
            set_species_homeworld = event_target:xhk_new_worldark_colony
        }
        every_owned_pop_species = {
            set_species_homeworld = event_target:xhk_new_worldark_colony
        }
        # giga_set_all_leaders_homeworld = {
        #     planet = event_target:xhk_new_worldark_colony
        # }

        #init mission setup fix
        capital_scope = {
            if = {
                limit = { owner = { has_country_flag = checklist_shield_realignment }}
                add_modifier = { modifier = checklist_unaligned_shields }
            }
            if = {
                limit = { owner = { has_country_flag = checklist_sensors_calibration }}
                add_modifier = { modifier = checklist_sensor_calibration }
            }
            if = {
                limit = { owner = { has_country_flag = checklist_flush_coolant }}
                add_modifier = { modifier = checklist_corrosive_coolant }
            }
            if = {
                limit = { owner = { has_country_flag = checklist_cluttered_hangars }}
                while = {
                    count = 2
                    add_deposit = d_cluttered_hangar_bay
                }
            }
            if = {
                limit = { owner = { has_country_flag = checklist_decipher_signal }}
                add_modifier = {
                    modifier = checklist_replicating_signal
                    days = -1
                }
            }
        }
    }
}
