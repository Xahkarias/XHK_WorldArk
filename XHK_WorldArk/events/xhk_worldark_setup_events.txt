namespace = xhk_worldark_setup

# INITIAL SETUP
country_event = {
    id = xhk_worldark_setup.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        xhk_has_worldark_origin = yes
    }

    immediate = {
        give_technology = {
            tech = xhk_tech_worldark_1
            message = no
        }

        ################################################################################################################
        # old capital scope
        # all about moving the pops, leaders, etc
        ################################################################################################################
        capital_scope = {
            solar_system = {
                save_event_target_as = target_system
            }
            save_event_target_as = xhk_old_arkcap_colony

            ship = {
                save_event_target_as = xhk_old_arkcap_ship
                fleet = { save_event_target_as = xhk_old_arkcap_fleet }

                # make the new ship
                create_fleet = {
                    effect = {
                        set_owner = root
                        create_ship = {
                            name = random
                            random_existing_design = xhk_worldark_tier_1
                            graphical_culture = xhk_worldark
                            create_colony = yes
                            effect = {
                                save_event_target_as = xhk_new_worldark_ship
                            }
                        }
                        set_location = event_target:target_system
                    }
                }

                # # grab the governor
                colony = {
                    save_event_target_as = xhk_new_worldark_colony
                    leader = {
                        unassign_leader = this
                        save_event_target_as = xhk_worldark_governor
                    }
                }

                # grab the admiral
                fleet = {
                    leader = {
                        unassign_leader = this
                        save_event_target_as = xhk_worldark_admiral
                    }
                }


                event_target:xhk_new_worldark_ship = {
                    # put leaders in new ship
                    colony = {
                        assign_leader = event_target:xhk_worldark_governor

                        set_capital = yes
                    }
                    fleet = {
                        assign_leader = event_target:xhk_worldark_admiral
                        # set the fleet name
                        set_name = {
                            key = "xhk_worldark_new_fleet_name"
                            variable_string = "[xhk_old_arkcap_fleet.GetName]"
                        }
                    }

                    set_name = {
                        key = "xhk_worldark_new_capital_name"
                        variable_string = "[xhk_old_arkcap_colony.GetName]"
                    }
                }
            }

            #move the pops
            # this is from the old capital scope to the new one
            every_owned_pop_group = {
                resettle_pop_group = {
                    POP_GROUP = this
                    PLANET = root.capital_scope
                    PERCENTAGE = 100
                }
            }
        }

        ################################################################################################################
        # new capital scope
        # district / starbase / deposit setup
        ################################################################################################################
        capital_scope = {

            # HAVE TO DO THIS SHIT MANUALLY
            ################################################################################

            # basic resource init
            set_variable = {
                which = energy_districts
                value = 2
            }
            set_variable = {
                which = mining_districts
                value = 2
            }
            set_variable = {
                which = farming_districts
                value = 3
            }


            # from authority regular nomad
            if = {
                limit = {
                    owner? = {
                        is_regular_empire = yes
                        is_nomadic = yes
                    }
                }
                add_district = district_xhk_worldark_research
                add_district = district_xhk_worldark_foundry
                add_district = district_xhk_worldark_factory
                add_district = district_xhk_worldark_trade
                add_building = {
                    building = building_factory_1
                }
				if = {
					limit = { owner = { is_spiritualist = yes }}
					add_district = district_xhk_worldark_unity_spiritualist
				}
				else = {
					add_district = district_xhk_worldark_unity
				}
            }
            # from authorithy hive nomad
            if = {
                limit = {
                    owner? = {
                        is_hive_empire = yes
                        is_wilderness_empire = no
                        is_nomadic = yes
                    }
                }
                if = {
                    limit = {
                        owner = {
                            NOT = { has_origin = origin_progenitor_hive }
                        }
                    }
                    add_building = building_spawning_pool
                }
                add_building = building_hive_warren

                add_district = district_xhk_worldark_research
                add_district = district_xhk_worldark_unity_hive
                add_district = district_xhk_worldark_foundry
                add_district = district_xhk_worldark_foundry
                add_district = district_xhk_worldark_trade

            }
            # from machine auth
            if = {
                limit = {
                    owner? = {
                        is_machine_empire = yes
                        is_nomadic = yes
                    }
                }
                add_building = building_machine_assembly_plant
                add_building = building_drone_storage

                add_district = district_xhk_worldark_research

                add_district = district_xhk_worldark_foundry
                add_district = district_xhk_worldark_trade

                if = {
                    limit = {
                        owner = {
                            OR = {
                                has_valid_civic = civic_machine_servitor
                                has_valid_civic = civic_machine_obsessional_directive
                            }
                        }
                    }
                    add_district = district_xhk_worldark_factory
					add_district = district_xhk_worldark_unity_servitor
                }
                else = {
					add_district = district_xhk_worldark_unity_machine
                    add_district = district_xhk_worldark_foundry
                }


                set_variable = {
                    which = energy_districts
                    value = 3
                }
                set_variable = {
                    which = mining_districts
                    value = 4
                }
                set_variable = {
                    which = farming_districts
                    value = 0
                }
            }

            ## Species
            inline_script = "game_start/species_lithoids"
            inline_script = "game_start/species_infernals"
            inline_script = "game_start/species_individual_machine"
            inline_script = "game_start/species_subspecies" ## Set subspecies flags

            # Shipsets
            inline_script = "game_start/shipsets"

            # NOT ADDING THIS BECAUSE IT CAN OVERFLOW THE START ZONE
            ## Ethic
            # inline_script = "game_start/ethic_spiritualist"
            # inline_script = "game_start/ethic_materialist"
            # inline_script = "game_start/ethic_pacifist"
            # inline_script = "game_start/ethic_militarist"
            # inline_script = "game_start/ethic_egalitarian"
            # inline_script = "game_start/ethic_authoritarian"
            # inline_script = "game_start/ethic_xenophile"
            # inline_script = "game_start/ethic_xenophobe"

            ##Civics
            inline_script = "game_start/civics"

            # Tankbound civic (removes every farming districts so we want to do that at the end)
            inline_script = "game_start/civic_tankbound"

            # energy
            if = {
                limit = {
                    is_planet_class = pc_ark
                }
                if = {
                    limit = {
                        check_variable = {
                            which = energy_districts
                            value > 2
                        }
                    }
                    set_variable = {
                        which = energy_districts
                        value = 2
                    }
                }
            }
            if = {
                limit = {
                    check_variable = {
                        which = energy_districts
                        value >= 1
                    }
                }
                add_district_and_planet_size_if_needed_effect_amount = {
                    amount = energy_districts
                    district = district_xhk_worldark_generator
                }
            }
            # food or minerals
            if = {
                limit = {
                    check_variable = {
                        which = farming_districts
                        value >= 1
                    }
                }
                add_district = district_xhk_worldark_farming
                add_zone = {
                    district = district_xhk_worldark_farming
                    zone = zone_xhk_worldark_farming
                }
            }
            else = {
                add_district = district_xhk_worldark_minerals
            }

            # Clear variables and flags
            clear_variable = energy_districts
            clear_variable = mining_districts
            clear_variable = farming_districts

            # AT THE END IF NEEDED
            upgrade_colony_shelter = yes

            #kill the extra pops made by moving the ship
            while = {
                count = 10
                random_owned_pop_group = {
                    limit = {
                        is_same_species = owner.species
                    }
                    kill_single_pop = yes
                }
            }

            ################################################################################
            #put in the old module / systems
            carrier = {
                initialize_arkship_starbase_effect = yes
            }

			#give the initial variables
			carrier = {
				set_variable = { which = xhk_worldark_science_upgrade_level value = 0 }
				set_variable = { which = xhk_worldark_military_upgrade_level value = 0 }
				set_variable = { which = xhk_worldark_civilian_upgrade_level value = 0 }
			}
			if = {
				limit = { owner = { has_country_flag = starting_civilian_arkship } }
				set_variable = { which = xhk_worldark_civilian_upgrade_level value = 1 }
			}
			else_if = {
				limit = { owner = { has_country_flag = starting_science_arkship } }
				set_variable = { which = xhk_worldark_science_upgrade_level value = 1 }
			}
			else_if = {
				limit = { owner = { has_country_flag = starting_military_arkship } }
				set_variable = { which = xhk_worldark_military_upgrade_level value = 1 }
			}

            ################################################################################
            # control deposit
            add_deposit = d_xhk_worldark_base

            # pop deposit
            if = {
                limit = {
                    owner  = { is_machine_empire = yes }
                }
                add_deposit = d_xhk_worldark_pop_machine
            }
            else_if = {
                limit = {
                    owner  = { is_hive_empire = yes }
                }
                add_deposit = d_xhk_worldark_pop_hive
            }
            else = {
                add_deposit = d_xhk_worldark_pop_standard
            }

            # var setup
            set_variable = {
                which = xhk_worldark_expansions
                value = 0
            }
            set_variable = {
                which = xhk_worldark_logistics
                value = 0
            }
            add_deposit = d_xhk_worldark_expansion_tracker
        }

        ################################################################################################################
        # MISC FIXES
        ################################################################################################################
        # homeworld fix
		every_owned_species = {
			set_species_homeworld = event_target:xhk_new_worldark_colony.ship
		}
        # every_owned_pop_species = {
        #     set_species_homeworld = event_target:xhk_new_worldark_colony
        # }
        # giga_set_all_leaders_homeworld = {
        #     planet = event_target:xhk_new_worldark_colony
        # }

		## the district trait
		every_owned_pop_group = {
			limit = { has_trait = trait_recursive_learners	}
			modify_species = {
				species = this
				remove_trait = trait_recursive_learners
				add_trait = trait_xhk_worldark_recursive_learners
				add_traits_at_start_of_list = yes
			}
		}
		every_owned_pop_group = {
			limit = { has_trait = trait_structural_awareness	}
			modify_species = {
				species = this
				remove_trait = trait_structural_awareness
				add_trait = trait_xhk_worldark_structural_awareness
				add_traits_at_start_of_list = yes
			}
		}

        #init mission setup fix
        capital_scope = {
            if = {
                limit = { owner = { has_country_flag = checklist_shield_realignment }}
                add_modifier = { modifier = checklist_unaligned_shields }
            }
            if = {
                limit = { owner = { has_country_flag = checklist_sensors_calibration }}
                add_modifier = { modifier = checklist_sensor_calibration }
            }
            if = {
                limit = { owner = { has_country_flag = checklist_flush_coolant }}
                add_modifier = { modifier = checklist_corrosive_coolant }
            }
            if = {
                limit = { owner = { has_country_flag = checklist_cluttered_hangars }}
                while = {
                    count = 2
                    add_deposit = d_cluttered_hangar_bay
                }
            }
            # checklist_decipher_signal
        }
        # ascensionists fix
        if = {
            limit = {
                is_ascensionist_empire = yes
            }
            every_owned_planet = {
                set_planetary_ascension_tier = 1
            }
        }
        # galvanic fix
        if = {
            limit = {
                is_galvanic_empire = yes
                capital_scope = {
                    NOT = { has_deposit = d_galvanic_living_metal }
                }
            }
            capital_scope = {
                add_deposit = d_galvanic_living_metal
            }
        }

		# remove zone if its there
		capital_scope = {
			if = {
				limit = {
					has_zone = { zone = zone_urban }
				}
				remove_zone = {
					district = district_xhk_worldark_city
					zone = zone_urban
				}
			}
		}

		# graphical culture
		event_target:xhk_new_worldark_ship = {
			set_graphical_culture = military_arkship
		}

        ################################################################################################################
        # setup docs
        ################################################################################################################
        begin_event_chain = {
            event_chain = xhk_worldark_docs_chain
        }
        create_point_of_interest = {
            id = xhk_worldark_docs_chain.1
            name = "xhk_worldark_docs_expansion"
            desc = "xhk_worldark_docs_expansion_desc"
            event_chain = "xhk_worldark_docs_chain"
            #location = capital_scope
        }
        create_point_of_interest = {
            id = xhk_worldark_docs_chain.2
            name = "xhk_worldark_docs_population"
            desc = "xhk_worldark_docs_population_desc"
            event_chain = "xhk_worldark_docs_chain"
            #location = capital_scope
        }
        create_point_of_interest = {
            id = xhk_worldark_docs_chain.3
            name = "xhk_worldark_docs_districts"
            desc = "xhk_worldark_docs_districts_desc"
            event_chain = "xhk_worldark_docs_chain"
            #location = capital_scope
        }
        create_point_of_interest = {
            id = xhk_worldark_docs_chain.4
            name = "xhk_worldark_docs_ship_levels"
            desc = "xhk_worldark_docs_ship_levels_desc"
            event_chain = "xhk_worldark_docs_chain"
            #location = capital_scope
        }
		create_point_of_interest = {
			id = xhk_worldark_docs_chain.5
			name = "xhk_worldark_docs_ship_upgrade"
			desc = "xhk_worldark_docs_ship_upgrade_desc"
			event_chain = "xhk_worldark_docs_chain"
			#location = capital_scope
		}
        create_point_of_interest = {
            id = xhk_worldark_docs_chain.6
            name = "xhk_worldark_docs_special_civic"
            desc = "xhk_worldark_docs_special_civic_desc"
            event_chain = "xhk_worldark_docs_chain"
            #location = capital_scope
        }
		create_point_of_interest = {
			id = xhk_worldark_docs_chain.7
			name = "xhk_worldark_docs_mega"
			desc = "xhk_worldark_docs_mega_desc"
			event_chain = "xhk_worldark_docs_chain"
			#location = capital_scope
		}
        create_point_of_interest = {
            id = xhk_worldark_docs_chain.8
            name = "xhk_worldark_docs_crossmod"
            desc = "xhk_worldark_docs_crossmod_desc"
            event_chain = "xhk_worldark_docs_chain"
            #location = capital_scope
        }
        start_situation = {
            type = xhk_worldark_situation_main
            target = capital_scope
        }
    }
}
