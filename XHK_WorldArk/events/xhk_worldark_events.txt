namespace = xhk_worldark

# Fired by xhk_on_size_harvest
carrier_event = {
    id = xhk_worldark.1
    hide_window = yes
    is_triggered_only = yes

    immediate = { # TARGET PLANET SCOPE
        from = {
            every_owned_ship = {
                save_event_target_as = arkship
            }
            if = {
                limit = {
                    event_target:arkship.starbase = {
                        has_starbase_size >= xhk_worldark_tier_5
                    }
                }
                event_target:arkship.colony = {
                    change_variable = {
                        which = xhk_worldark_expansions
                        value = 2
                    }
                    add_deposit = d_xhk_worldark_harvest_size
                    add_deposit = d_xhk_worldark_harvest_size
                }
            }
            else = {
                event_target:arkship.colony = {
                    change_variable = {
                        which = xhk_worldark_expansions
                        value = 1
                    }
                    add_deposit = d_xhk_worldark_harvest_size
                }
            }
        }

        # from.starbase.ship.carrier = { # ARKSHIP SCOPE
        #     add_deposit = d_xhk_worldark_harvest_size
        # }

        # forever
        add_modifier = {
            modifier = xhk_size_harvested_already
        }

        if = {
            limit = {
                planet_size > 5
            }
            change_planet_size = -1
        }

        # Always decrease opinion of system owner
        space_owner? = {
            add_opinion_modifier = {
                modifier = opinion_arkship_strip_mine
                who = event_target:arkship.controller
            }
        }
    }
}


# EAT THE OTHER SHIP
# triggered monthly
country_event = {
    id = xhk_worldark.100
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_nomadic = yes
        xhk_has_worldark_origin = yes
        num_owned_colonies > 1
    }

    immediate = {
        while = {
            count = value:xhk_worldark_num_arkships_minus_1
            capital_scope = {
                change_variable = {
                    which = xhk_worldark_expansions
                    value = 4
                }
                add_deposit = d_xhk_worldark_consume_ark_size
            }
        }

        every_owned_pop_group = {
            limit = {
                this.colony = {
                    NOT = { is_same_value = root.capital_scope }
                }
            }
            resettle_pop_group = {
                POP_GROUP = this
                PLANET = root.capital_scope
                PERCENTAGE = 100
            }
        }

        create_message = {
            type = XHK_TOAST_CONSUME_ARKSHIP_COMPLETE
            localization =  XHK_TOAST_CONSUME_ARKSHIP_COMPLETE_DESC
            days = @toast_message_days
            target = capital_scope
        }
    }
}
