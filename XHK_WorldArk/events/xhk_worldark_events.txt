# Fired by xhk_on_size_harvest
carrier_event = {
    id = xhk_worldark.1
    hide_window = yes
    is_triggered_only = yes

    immediate = { # TARGET PLANET SCOPE
        from = {
            every_owned_ship = {
                save_event_target_as = arkship
            }
			###### TIER 11 UPGRADE
			if = {
				limit = {
					event_target:arkship.starbase = {
						has_starbase_size >= xhk_worldark_tier_11
					}
				}
				event_target:arkship.colony = {
					change_variable = {
						which = xhk_worldark_expansions
						value = @xhk_worldark_harvest_size_vlarge
					}
					while = {
						count = @xhk_worldark_harvest_size_vlarge
						add_deposit = d_xhk_worldark_harvest_size
					}
				}
				root = {
					if = {
						limit = {
							planet_size > @xhk_worldark_harvest_size_vlarge
						}
						change_planet_size = @[xhk_worldark_harvest_size_vlarge * -1]
					}
					else = {
						set_planet_size = 1
					}
				}
			}
			###### TIER 5 UPGRADE
            else_if = {
                limit = {
                    event_target:arkship.starbase = {
                        has_starbase_size >= xhk_worldark_tier_5
                    }
                }
                event_target:arkship.colony = {
                    change_variable = {
                        which = xhk_worldark_expansions
                        value = @xhk_worldark_harvest_size_large
                    }
					while = {
						count = @xhk_worldark_harvest_size_large
						add_deposit = d_xhk_worldark_harvest_size
					}
                }
				root = {
					if = {
						limit = {
							planet_size > @xhk_worldark_harvest_size_large
						}
						change_planet_size = @[xhk_worldark_harvest_size_large * -1]
					}
					else = {
						set_planet_size = 1
					}
				}
            }
			###### NO UPGRADE
            else = {
                event_target:arkship.colony = {
                    change_variable = {
                        which = xhk_worldark_expansions
                        value = @xhk_worldark_harvest_size_small
                    }
					while = {
						count = @xhk_worldark_harvest_size_small
						add_deposit = d_xhk_worldark_harvest_size
					}
                }
				root = {
					if = {
						limit = {
							planet_size > @xhk_worldark_harvest_size_small
						}
						change_planet_size = @[xhk_worldark_harvest_size_small * -1]
					}
					else = {
						set_planet_size = 1
					}
				}
            }
            event_target:arkship.fleet = {
                # this tags the fleet, easier to reference than the ship or the colony
                set_timed_fleet_flag = { flag = xhk_expand_harvest_cooldown days = @xhk_worldark_harvest_cd }
            }

			#cooldown toaster
			owner = { country_event = { id = xhk_worldark.2 days = @xhk_worldark_harvest_cd } }
        }

        # from.starbase.ship.carrier = { # ARKSHIP SCOPE
        #     add_deposit = d_xhk_worldark_harvest_size
        # }

        # forever
        add_modifier = {
            modifier = xhk_size_harvested_already
        }

        # Always decrease opinion of system owner
        space_owner? = {
            add_opinion_modifier = {
                modifier = opinion_arkship_strip_mine
                who = event_target:arkship.controller
            }
        }
    }
}

# EAT THE OTHER SHIP
# triggered monthly
country_event = {
    id = xhk_worldark.100
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        is_nomadic = yes
        xhk_has_worldark_origin = yes
        num_owned_colonies > 1
    }

    immediate = {
        while = {
            count = value:xhk_worldark_num_arkships_minus_1
            capital_scope = {
                change_variable = {
                    which = xhk_worldark_expansions
                    value = 4
                }
                add_deposit = d_xhk_worldark_consume_ark_size
            }
        }

        every_owned_pop_group = {
            limit = {
                this.colony = {
                    NOT = { is_same_value = root.capital_scope }
                }
            }
            resettle_pop_group = {
                POP_GROUP = this
                PLANET = root.capital_scope
                PERCENTAGE = 100
            }
        }

        create_message = {
            type = XHK_TOAST_CONSUME_ARKSHIP_COMPLETE
            localization =  XHK_TOAST_CONSUME_ARKSHIP_COMPLETE_DESC
            days = @toast_message_days
            target = capital_scope
        }
    }
}

# triggered on killing another ship
# for the nanite spawn
country_event = {
	id = xhk_worldark.300
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		xhk_has_worldark_origin = yes
		fromfrom.starbase = {
			has_starbase_module = xhk_worldark_corespinal_nanite
		}
	}
	immediate = {
		FROMFROMFROM = { save_event_target_as = xhk_worldark_destroyed_ship }

		# increment the var
		FROMFROM = {
			change_variable = {
				which = xhk_worldark_nanite_num_killcap
				value = event_target:xhk_worldark_destroyed_ship.value:ship_naval_cap
			}
		}

		# spawn the fleet
		while = {
			limit = {
				FROMFROM = {
					check_variable = {
						which = xhk_worldark_nanite_num_killcap
						value >= 100
					}
				}
			}
			create_fleet = {
				name = "NAME_XHK_WORLDARK_NANITE_SPAWN"
				effect = {
					set_owner = root
					create_ship = {
						random_existing_design = nanite_interdictor
					}
					while = {
						count = 15
						create_ship = {
							random_existing_design = nanite_swarmer
						}
					}
					set_location = root.FromFrom
					set_fleet_stance = aggressive
					set_fleet_flag = xhk_worldark_nanite_core_spinal_fleet
				}
			}
			FROMFROM = {
				change_variable = {
					which = xhk_worldark_nanite_num_killcap
					value = -100
				}
			}
		}
	}
}

				}
			}
			destroy_fleet = this
		}
	}
}

# Starts the consume situation on first arrival, or resumes it if the fleet returns after cancelling.
# THIS = planet
# FROM = triggering fleet
planet_event = {
	id = xhk_worldark.2000
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_carrier_flag = being_devoured } }
			# First arrival: full initialisation
			set_timed_carrier_flag = {
				days = 360
				flag = recently_eaten_planet
			}
			set_carrier_flag = colony_event
			set_carrier_flag = being_devoured
			add_modifier = {
				modifier = xhk_worldark_being_devoured_modifier
				days = -1
			}
			export_trigger_value_to_variable = {
				trigger = planet_size
				variable = num_districts_xhk_worldark_consume
			}
			from.controller = {
				start_situation = {
					type = xhk_worldark_situation_planeteater
					target = root
				}
			}
		}
		else = {
			# Returning after cancellation: clear the grace period flag
			remove_carrier_flag = arkship_absent_grace
		}
		from = { set_fleet_flag = consuming_world_arkship@root }
	}
}

# THIS = situation, owner = consuming empire, target = planet
situation_event = {
	id = xhk_worldark.2002
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		# Consume tick: only when arkship is present and not on cooldown
		if = {
			limit = {
				target = {
					NOR = {
						has_carrier_flag = recently_eaten_planet
						has_carrier_flag = arkship_absent_grace
					}
				}
			}
			random_list = {
				10 = {
					modifier = {
						factor = 0
						target = { planet_size <= @xhk_worldark_dismantle_size_decrease }
					}
					target = {
						change_planet_size = @[xhk_worldark_dismantle_size_decrease * -1]
					}
					owner = {
						add_resource = {
							minerals = 4000
							mult = value:xhk_worldark_dismantle_res_boost
						}
						add_resource = {
							alloys = 4000
							mult = value:xhk_worldark_dismantle_res_boost
						}
						create_message = {
							type = XHK_MESSAGE_PLANETEATER_PROCESS
							localization = XHK_MESSAGE_PLANETEATER_PROCESS_DESC
							days = 30
							target = root.target
							variable = { type = name localization = PLANET scope = root.target }
						}
						save_event_target_as = xhk_worldark_consuming_empire

						# check if habitable
						if = {
							limit = {
								root.target = { colonizable_planet = yes }
							}
							every_country = {
								limit = {
									is_nomadic = no
									NOT = { is_same_value = event_target:xhk_worldark_consuming_empire }
									has_communications = event_target:xhk_worldark_consuming_empire
								}
								add_opinion_modifier = {
									modifier = opinion_xhk_worldark_planeteater_habitable
									who = event_target:xhk_worldark_consuming_empire
								}
							}
						}
						else = {
							every_country = {
								limit = {
									is_nomadic = no
									NOT = { is_same_value = event_target:xhk_worldark_consuming_empire }
									has_communications = event_target:xhk_worldark_consuming_empire
								}
								add_opinion_modifier = {
									modifier = opinion_xhk_worldark_planeteater_barren
									who = event_target:xhk_worldark_consuming_empire
								}
							}
						}


						capital_scope = {
							change_variable = {
								which = xhk_worldark_expansions
								value = 1
							}
							add_deposit = d_xhk_worldark_dismantle_size
						}
					}
				}
			}
			target = {
				set_timed_carrier_flag = { flag = recently_eaten_planet days = @xhk_worldark_dismantle_pulse_cooldown }
			}
		}
	}
}

situation_event = {
	id = xhk_worldark.2003
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		## opinion change
		if = {
			limit = {
				root.target = { colonizable_planet = yes }
			}
			every_country = {
				limit = {
					is_nomadic = no
					NOT = { is_same_value = event_target:xhk_worldark_consuming_empire }
					has_communications = event_target:xhk_worldark_consuming_empire
				}
				add_opinion_modifier = {
					modifier = opinion_xhk_worldark_planeteater_habitable
					who = event_target:xhk_worldark_consuming_empire
				}
			}
		}
		else = {
			every_country = {
				limit = {
					is_nomadic = no
					NOT = { is_same_value = event_target:xhk_worldark_consuming_empire }
					has_communications = event_target:xhk_worldark_consuming_empire
				}
				add_opinion_modifier = {
					modifier = opinion_xhk_worldark_planeteater_barren
					who = event_target:xhk_worldark_consuming_empire
				}
			}
		}
		# kill it
		target = {
			# just to be nice and clean
			remove_carrier_flag = being_devoured
			remove_carrier_flag = colony_event
			remove_carrier_flag = arkship_absent_grace
			remove_modifier = xhk_worldark_being_devoured_modifier
			remove_planet = yes
		}
		owner = {
			save_event_target_as = xhk_worldark_consuming_empire
			random_owned_fleet = {
				limit = { has_fleet_flag = consuming_world_arkship@root.target }
				remove_fleet_flag = consuming_world_arkship@root.target
				save_event_target_as = consuming_fleet
			}
			create_message = {
				type = XHK_MESSAGE_PLANETEATER_COMPLETE
				localization = XHK_MESSAGE_PLANETEATER_COMPLETE_DESC
				days = 30
				target = root.target
				variable = { type = name localization = PLANET scope = root.target }
				variable = { type = name localization = ARKSHIP scope = event_target:consuming_fleet }
			}
		}

		destroy_situation = this
	}
